cmake_minimum_required(VERSION 3.23)
project(DXVK LANGUAGES CXX)

set(DXVK_LIB_ROOT "${CMAKE_CURRENT_LIST_DIR}/lib")

if(APPLE)
	if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
		set(_dxvk_arch_dir "aarch64")
	else()
		message(FATAL_ERROR "DXVK-CMake: Unsupported macOS architecture '${CMAKE_SYSTEM_PROCESSOR}'.")
	endif()
	set(_dxvk_prefix "libdxvk_")
	set(_dxvk_suffix ".dylib")
elseif(WIN32)
	if(CMAKE_SIZEOF_VOID_P EQUAL 8)
		set(_dxvk_arch_dir "x64")
	else()
		set(_dxvk_arch_dir "x32")
	endif()
	set(_dxvk_prefix "")
	set(_dxvk_suffix ".dll")
elseif(UNIX)
	if(CMAKE_SIZEOF_VOID_P EQUAL 8)
		set(_dxvk_arch_dir "x64")
	else()
		set(_dxvk_arch_dir "x32")
	endif()
	set(_dxvk_prefix "libdxvk_")
	set(_dxvk_suffix ".so")
else()
	message(FATAL_ERROR "DXVK-CMake: Unsupported platform '${CMAKE_SYSTEM_NAME}'.")
endif()

function(dxvk_add_imported target_name lib_stem)
	set(_lib_path "${DXVK_LIB_ROOT}/${_dxvk_arch_dir}/${_dxvk_prefix}${lib_stem}${_dxvk_suffix}")
	if(NOT EXISTS "${_lib_path}")
		message(FATAL_ERROR "DXVK-CMake: Missing binary for ${target_name}: ${_lib_path}")
	endif()

	add_library(${target_name} UNKNOWN IMPORTED GLOBAL)
	set_target_properties(${target_name} PROPERTIES
		IMPORTED_LOCATION "${_lib_path}"
	)
	target_include_directories(${target_name} INTERFACE "${CMAKE_CURRENT_LIST_DIR}/include/dxvk")
endfunction()

dxvk_add_imported(DXVK::D3D8 d3d8)
dxvk_add_imported(DXVK::D3D9 d3d9)
# DXVK ships d3d10core, use it for D3D10 linking.
dxvk_add_imported(DXVK::D3D10 d3d10core)
dxvk_add_imported(DXVK::D3D11 d3d11)
dxvk_add_imported(DXVK::DXGI dxgi)

add_library(D3D8 ALIAS DXVK::D3D8)
add_library(D3D9 ALIAS DXVK::D3D9)
add_library(D3D10 ALIAS DXVK::D3D10)
add_library(D3D11 ALIAS DXVK::D3D11)
add_library(DXGI ALIAS DXVK::DXGI)
